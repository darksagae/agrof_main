import os

print("🔍 Checking PlantVillage for Beans...")
print("="*60)

plantvillage_base = '/content/datasets/plantvillage'

# Search for Bean/Beans in PlantVillage
bean_classes = []
for root, dirs, files in os.walk(plantvillage_base):
    for dir_name in dirs:
        if 'Bean' in dir_name or 'bean' in dir_name:
            source_path = os.path.join(root, dir_name)
            try:
                image_count = len([f for f in os.listdir(source_path) 
                                  if f.lower().endswith(('.jpg', '.jpeg', '.png'))])
                if image_count > 0:
                    bean_classes.append((dir_name, image_count))
                    print(f"🌱 Found: {dir_name} ({image_count} images)")
            except:
                pass

if bean_classes:
    total_bean_images = sum(c[1] for c in bean_classes)
    print(f"\n✅ Found {len(bean_classes)} Bean classes with {total_bean_images:,} images!")
else:
    print("❌ No Bean classes found in PlantVillage")
