import os
import shutil

print("📁 Organizing all datasets into /content/agrof_dataset...")
print("="*60)

# Create main directory
os.makedirs('/content/agrof_dataset', exist_ok=True)

# Function to copy classes
def copy_classes(source_base, crop_prefix):
    """Copy all disease classes from source to agrof_dataset"""
    copied = 0
    
    if not os.path.exists(source_base):
        print(f"⚠️  {source_base} not found, skipping...")
        return copied
    
    for root, dirs, files in os.walk(source_base):
        for dir_name in dirs:
            source_path = os.path.join(root, dir_name)
            
            # Count images
            try:
                image_files = [f for f in os.listdir(source_path) 
                              if f.lower().endswith(('.jpg', '.jpeg', '.png', '.JPG', '.PNG'))]
                
                if len(image_files) > 50:  # Only copy if has sufficient images
                    # Clean name
                    clean_name = dir_name.replace('___', '_').replace('__', '_').replace(' ', '_')
                    
                    # Add prefix if not present
                    if not clean_name.startswith(crop_prefix):
                        clean_name = f'{crop_prefix}_{clean_name}'
                    
                    dest_path = f'/content/agrof_dataset/{clean_name}'
                    
                    # Copy if doesn't exist
                    if not os.path.exists(dest_path):
                        print(f"📦 {clean_name}: {len(image_files)} images")
                        shutil.copytree(source_path, dest_path)
                        copied += 1
            except Exception as e:
                pass
    
    return copied

# 1. Copy Beans (from wherever it downloaded)
print("\n🌱 ORGANIZING BEANS...")
beans_copied = 0
beans_copied += copy_classes('/content/datasets/beans', 'Beans')

import os

print("🔍 Final dataset verification...")
print("="*60)

crops = {
    'Beans': [],
    'Coffee': [],
    'Tomato': [],
    'Potato': [],
    'Pepper': []
}

# Count all classes
for class_name in sorted(os.listdir('/content/agrof_dataset')):
    class_path = f'/content/agrof_dataset/{class_name}'
    if os.path.isdir(class_path):
        count = len([f for f in os.listdir(class_path) 
                    if f.lower().endswith(('.jpg', '.jpeg', '.png'))])
        
        # Categorize
        if 'Bean' in class_name:
            crops['Beans'].append((class_name, count))
        elif 'Coffee' in class_name:
            crops['Coffee'].append((class_name, count))
        elif 'Tomato' in class_name:
            crops['Tomato'].append((class_name, count))
        elif 'Potato' in class_name:
            crops['Potato'].append((class_name, count))
        elif 'Pepper' in class_name:
            crops['Pepper'].append((class_name, count))

# Display
for crop, classes_list in crops.items():
    if classes_list:
        total = sum(c[1] for c in classes_list)
        icon = {'Beans': '🌱', 'Coffee': '☕', 'Tomato': '🍅', 
               'Potato': '🥔', 'Pepper': '🌶️'}[crop]
        print(f"\n{icon} {crop}: {len(classes_list)} classes, {total:,} images")
        for class_name, count in classes_list:
            disease = class_name.split('_', 1)[1] if '_' in class_name else class_name
            print(f"   ├─ {disease}: {count:,} images")

# Summary
total_classes = sum(len(c) for c in crops.values())
total_images = sum(sum(cls[1] for cls in c) for c in crops.values())

print(f"\n{'='*60}")
print(f"✅ READY FOR TRAINING!")
print(f"📊 Total: {total_classes} classes, {total_images:,} images")
print("="*60)

if total_classes >= 10:
    print("\n🎉 Dataset is ready! Proceed with training!")
    print("📋 Next: Run training cells from COMPLETE_TRAINING_CODE_FOR_COLAB.txt")
else:
    print(f"\n⚠️  Only {total_classes} classes found")
